cmake_minimum_required(VERSION 3.25)
project(prumo VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SQLite3 REQUIRED)
if(TARGET SQLite3::SQLite3)
  set(PRUMO_SQLITE_TARGET SQLite3::SQLite3)
else()
  set(PRUMO_SQLITE_TARGET SQLite::SQLite3)
endif()
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(prumo src/http/main.cpp src/storage/store.cpp src/domain/model.cpp src/application/service.cpp)
target_include_directories(prumo PRIVATE include)
target_include_directories(prumo SYSTEM PRIVATE third_party)
target_link_libraries(prumo PRIVATE ${PRUMO_SQLITE_TARGET} OpenSSL::Crypto Threads::Threads)
target_compile_options(prumo PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(prumo PRIVATE PRUMO_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME invariants COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:prumo>)
set_tests_properties(invariants PROPERTIES TIMEOUT 90)
